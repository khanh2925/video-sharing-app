export default function RootNavigator() {
  const { user, isBootstrapping } = useAuth();
  const rootNav = useContext(NavigationContainerRefContext);
  // user từ có → null: xoá hết các màn đang mở, về lại MainTabs
  const wasLoggedIn = useRef(false);
  useEffect(() => {
    if (wasLoggedIn.current && !user && rootNav?.isReady()) {
      rootNav.reset({ index: 0, routes: [{ name: 'MainTabs' }] });
    }
    wasLoggedIn.current = !!user;
  }, [user, rootNav]);
  if (isBootstrapping) {
    return <Loading />;
  }

export function getErrorMessage(error: unknown): string {
  if (axios.isAxiosError<ApiErrorBody>(error)) {
    const message = error.response?.data?.message;
    if (message) return message;
    if (error.code === 'ECONNABORTED') return 'Máy chủ phản hồi quá lâu, thử lại sau';
    if (!error.response) return 'Không kết nối được máy chủ. Kiểm tra mạng hoặc EXPO_PUBLIC_API_URL';
    return `Có lỗi xảy ra (mã ${error.response.status})`;
  }
  if (error instanceof Error && error.message) return error.message;
  return 'Có lỗi xảy ra, thử lại sau';
}

/** 950 → "950", 8912 → "8.912", 12400 → "12,4N", 1200000 → "1,2Tr" */
export function formatCount(n: number): string {
  if (n < 10_000) return n.toLocaleString('vi-VN');
  if (n < 1_000_000) return `${trimDecimal(n / 1_000)}N`;
  return `${trimDecimal(n / 1_000_000)}Tr`;
}
function trimDecimal(value: number): string {
  return (Math.floor(value * 10) / 10).toString().replace('.', ',');
}
/** 28 → "0:28", 75 → "1:15" */
export function formatDuration(totalSec: number): string {
  const min = Math.floor(totalSec / 60);
  const sec = Math.floor(totalSec % 60);
  return `${min}:${sec.toString().padStart(2, '0')}`;
}

/** "2026-10-25T10:00:00" → "Vừa xong" / "5 phút trước" / "3 giờ trước" / "2 ngày trước" / "25/10/2026" */
export function timeAgo(iso: string): string {
  const diffSec = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000);
  if (diffSec < 60) return 'Vừa xong';
  if (diffSec < 3600) return `${Math.floor(diffSec / 60)} phút trước`;
  if (diffSec < 86400) return `${Math.floor(diffSec / 3600)} giờ trước`;
  if (diffSec < 7 * 86400) return `${Math.floor(diffSec / 86400)} ngày trước`;
  return new Date(iso).toLocaleDateString('vi-VN');
}

/**
 * Tạo "file" để append vào FormData khi upload (React Native gửi file theo kiểu {uri, name, type}).
 *   const form = new FormData();
 *   form.append('file', fileFromUri(asset.uri, asset.mimeType));
 *   form.append('title', title);
 *   await postMultipart<Video>('/videos', form);
 */
export function fileFromUri(uri: string, mimeType?: string | null) {
  const name = uri.split('/').pop() ?? 'upload';
  // React Native chấp nhận object này trong FormData; kiểu TS mặc định chỉ biết Blob nên ép kiểu
  return { uri, name, type: mimeType ?? guessMimeType(uri) } as unknown as Blob;
}

  // useAudioPlayer tự giải phóng khi rời màn hình
  const player = useAudioPlayer(uri ? { uri } : null);
